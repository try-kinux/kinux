start_musl
version=1.2.5
description=The musl C library - lightweight, fast, and correct
homepage=https://musl.libc.org/
license=MIT
maintainer=Kinux Team <try-kinux@proton.me>
essential=yes
fetch=https://musl.libc.org/releases/musl-1.2.5.tar.gz

[instructions]
./configure --prefix=/usr --syslibdir=/usr/lib
make -j$(nproc)
make install
ln -sf /usr/lib/libc.so /usr/lib/ld-musl-x86_64.so.1
end_musl

start_busybox
version=1.36.1
description=The Swiss Army Knife of embedded Linux
homepage=https://busybox.net/
license=GPL-2.0-only
maintainer=Kinux Team <try-kinux@proton.me>
essential=yes
fetch=https://busybox.net/downloads/busybox-1.36.1.tar.bz2

[dependencies]
musl

[instructions]
make defconfig
sed -i 's/# CONFIG_STATIC is not set/CONFIG_STATIC=y/' .config
make -j$(nproc)
make install CONFIG_PREFIX=/usr
end_busybox
